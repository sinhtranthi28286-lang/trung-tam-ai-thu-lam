function aiStaffForArea(area){
 const exact=TT_STAFF_WORK.filter(x=>x.area===area);
 return exact.length?exact:TT_STAFF_WORK.filter(x=>String(x.area||'').toLowerCase().includes(String(area||'').toLowerCase().replace('phụ trách ','')));
}
function splitAITasks(text){
 let raw=String(text||'').replace(/\r/g,'\n').replace(/\n{2,}/g,'\n').trim();
 if(!raw)return [];
 let lines=raw.split('\n').map(x=>x.trim()).filter(Boolean);
 let candidates=[];
 lines.forEach(line=>{
   let cleaned=line.replace(/^[\-\+\•\*\d\.\)\(]+\s*/,'').trim();
   if(cleaned.length<18)return;
   if(/^(căn cứ|kính gửi|nơi nhận|đảng cộng sản|ủy ban|ban xây dựng|số:|ngày)/i.test(cleaned))return;
   if(/(giao|đề nghị|yêu cầu|thực hiện|tham mưu|rà soát|tổng hợp|báo cáo|triển khai|hoàn thành|phối hợp|ban hành|xây dựng|chuẩn bị|đôn đốc)/i.test(cleaned)){
     candidates.push(cleaned)
   }
 });
 if(!candidates.length){
   const sentences=raw.split(/(?<=[\.\;\:])\s+/).map(x=>x.trim()).filter(x=>x.length>30);
   candidates=sentences.filter(x=>/(giao|đề nghị|yêu cầu|thực hiện|tham mưu|rà soát|tổng hợp|báo cáo|triển khai|hoàn thành|phối hợp|ban hành|xây dựng|chuẩn bị|đôn đốc)/i.test(x));
 }
 return [...new Set(candidates)].slice(0,30);
}
function classifyAITask(task){
 let t=task.toLowerCase(),best={area:'',score:0};
 AI_RULES.forEach(r=>{let score=r.keys.reduce((n,k)=>n+(t.includes(k)?1:0),0);if(score>best.score)best={area:r.area,score}});
 if(!best.area){
   const staffHit=TT_STAFF_WORK.find(x=>String(x.area||'').split(',').some(k=>k.length>4&&t.includes(k.toLowerCase().trim())));
   if(staffHit)best={area:staffHit.area,score:1}
 }
 return best
}
function inferAIProduct(task){
 const t=task.toLowerCase();
 if(t.includes('báo cáo'))return 'Báo cáo';
 if(t.includes('quyết định')||t.includes('ban hành quyết định'))return 'Dự thảo/Quyết định';
 if(t.includes('kế hoạch'))return 'Kế hoạch';
 if(t.includes('công văn'))return 'Công văn';
 if(t.includes('tờ trình'))return 'Tờ trình';
 if(t.includes('tổng hợp'))return 'Bảng/Danh sách tổng hợp';
 if(t.includes('rà soát'))return 'Kết quả rà soát/Tổng hợp';
 if(t.includes('tham mưu'))return 'Dự thảo văn bản tham mưu';
 return '[CẦN LÃNH ĐẠO XÁC ĐỊNH]'
}
function inferAIDeadline(task,fullText){
 const txt=task+' '+fullText;
 let m=txt.match(/(?:trước|chậm nhất|hoàn thành trước)\s+(?:ngày\s*)?(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{4})/i);
 if(m)return `${m[3]}-${String(m[2]).padStart(2,'0')}-${String(m[1]).padStart(2,'0')}`;
 let m2=txt.match(/ngày\s+(\d{1,2})\s+tháng\s+(\d{1,2})\s+năm\s+(\d{4})/i);
 if(m2)return `${m2[3]}-${String(m2[2]).padStart(2,'0')}-${String(m2[1]).padStart(2,'0')}`;
 return ''
}
function sourceExcerpt(task,text){
 let idx=text.toLowerCase().indexOf(task.toLowerCase().slice(0,30));
 if(idx<0)return task.slice(0,240);
 return text.slice(Math.max(0,idx-80),Math.min(text.length,idx+task.length+120)).replace(/\n+/g,' ')
}
async function readAIDocument(){
 const f=aiDocFile.files?.[0];
 if(!f)return alert('Chọn file PDF hoặc DOCX.');
 aiReadStatus.textContent='Đang đọc văn bản...';
 try{
   let text='';
   if(f.name.toLowerCase().endsWith('.pdf')){
     if(!window.pdfjsLib)throw new Error('Chưa tải được thư viện đọc PDF.');
     pdfjsLib.GlobalWorkerOptions.workerSrc='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
     const buf=await f.arrayBuffer(),pdf=await pdfjsLib.getDocument({data:buf}).promise;
     let parts=[];
     for(let i=1;i<=pdf.numPages;i++){
       const page=await pdf.getPage(i),content=await page.getTextContent();
       parts.push(content.items.map(x=>x.str).join(' '))
     }
     text=parts.join('\n')
   }else if(f.name.toLowerCase().endsWith('.docx')){
     if(!window.mammoth)throw new Error('Chưa tải được thư viện đọc Word.');
     const buf=await f.arrayBuffer(),res=await mammoth.extractRawText({arrayBuffer:buf});
     text=res.value||''
   }else throw new Error('Bản thử nghiệm hỗ trợ PDF và DOCX.');
   aiDocText.value=text.trim();
   aiReadStatus.innerHTML=`<b>Đã đọc:</b> ${ce(f.name)} • ${text.length.toLocaleString('vi-VN')} ký tự.`;
 }catch(e){console.error(e);aiReadStatus.textContent='Không đọc được file: '+e.message}
}
function analyzeAIDocument(){
 const text=aiDocText.value.trim();if(!text)return alert('Chưa có nội dung văn bản để phân tích.');
 const tasks=splitAITasks(text);
 AI_PROPOSALS=tasks.map((task,i)=>{
   const cls=classifyAITask(task),people=aiStaffForArea(cls.area),lead=people[0]?.name||'',coord=people.slice(1).map(x=>x.name).join(', ');
   return {id:Date.now()+i,selected:true,task,area:cls.area||'[CẦN XÁC ĐỊNH]',lead,coord,product:inferAIProduct(task),due:inferAIDeadline(task,text),source:sourceExcerpt(task,text),review:!cls.area||!lead}
 });
 renderAIProposals();
 aiReadStatus.innerHTML=`Đã phân tích văn bản. Phát hiện <b>${AI_PROPOSALS.length}</b> nội dung có dấu hiệu là nhiệm vụ cần theo dõi.`
}

async function recoverLegacyDirectives(){
 try{
  if(typeof sb==='undefined')return;
  ensureDirectives?.();
  const {data,error}=await sb.from('xa_directives').select('*').order('id',{ascending:true});if(error||!data?.length)return;
  const existing=new Set((CTRL.directives||[]).map(x=>String(x.no||x.document_no||'').trim()+'|'+String(x.title||'').trim()));
  let changed=false;
  for(const d of data){
   const k=String(d.document_no||'').trim()+'|'+String(d.title||'').trim();if(existing.has(k))continue;
   CTRL.directives.push({id:'legacy-'+d.id,no:d.document_no||'',date:d.issued_date||'',title:d.title||'',area:d.category||'Văn bản chỉ đạo',summary:d.summary||'',fileUrl:d.file_url||'',source:'xa_directives'});existing.add(k);changed=true;
  }
  if(changed){localStorage.setItem(CTRLKEY,JSON.stringify(CTRL));renderDirectives?.();renderDocumentLookup?.()}
 }catch(e){console.warn('Legacy directive recovery skipped',e)}
}

async function readAIDocument(){
 const f=aiDocFile.files?.[0];
 if(!f)return alert('Chọn file PDF hoặc DOCX.');
 aiReadStatus.textContent='Đang đọc văn bản...';
 try{
   let text='';
   if(f.name.toLowerCase().endsWith('.pdf')){
     if(!window.pdfjsLib)throw new Error('Chưa tải được thư viện đọc PDF.');
     pdfjsLib.GlobalWorkerOptions.workerSrc='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
     const buf=await f.arrayBuffer(),pdf=await pdfjsLib.getDocument({data:buf}).promise;
     let parts=[];
     for(let i=1;i<=pdf.numPages;i++){
       const page=await pdf.getPage(i),content=await page.getTextContent();
       parts.push(content.items.map(x=>x.str).join(' '))
     }
     text=parts.join('\n')
   }else if(f.name.toLowerCase().endsWith('.docx')){
     if(!window.mammoth)throw new Error('Chưa tải được thư viện đọc Word.');
     const buf=await f.arrayBuffer(),res=await mammoth.extractRawText({arrayBuffer:buf});
     text=res.value||''
   }else throw new Error('Bản thử nghiệm hỗ trợ PDF và DOCX.');
   aiDocText.value=text.trim();
   aiReadStatus.innerHTML=`<b>Đã đọc:</b> ${ce(f.name)} • ${text.length.toLocaleString('vi-VN')} ký tự.`;
 }catch(e){console.error(e);aiReadStatus.textContent='Không đọc được file: '+e.message}
}
